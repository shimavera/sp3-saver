# IG Saver Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extensão de Chrome (uso pessoal, instalação descompactada) que baixa foto, vídeo, carrossel, story, destaque e avatar do Instagram com um clique num botão flutuante sobre a mídia.

**Architecture:** Manifest V3 sem build step. Um content script roda no instagram.com, detecta mídia via `MutationObserver` e injeta um botão em Shadow DOM. A URL real do arquivo vem de um resolver em camadas (endpoint interno do Instagram usando a sessão logada → leitura do DOM → erro visível). O download é executado pelo service worker via `chrome.downloads`.

**Tech Stack:** JavaScript ESM puro, zero dependências de runtime. Testes com o runner nativo do Node (`node --test`), só sobre lógica pura (sem DOM, sem rede).

**Spec:** `docs/superpowers/specs/2026-08-15-ig-saver-design.md`

## Global Constraints

- Node 18+ para rodar os testes (`node --test`). Nenhuma dependência npm de runtime; `package.json` existe apenas para marcar `"type": "module"` e o script de teste.
- Manifest V3. Permissões exatas: `downloads` e `host_permissions` para `https://www.instagram.com/*`, `https://*.cdninstagram.com/*`, `https://*.fbcdn.net/*`. Nada além disso.
- Nenhuma requisição para fora do domínio do Instagram/CDN. Sem analytics, sem telemetria, sem `storage` remoto.
- Todo módulo em `src/lib/` é puro: não toca `document`, `window`, `fetch` nem `chrome`. É o que os testes cobrem.
- Nome do arquivo salvo = nome original do CDN, sanitizado. Sem subpasta.
- Story: nunca chamar endpoint de `seen`/`view`. Só leitura.
- Mensagens de erro visíveis ao usuário em português.
- App id usado nos headers: `936619743392459` (o mesmo que o site usa no navegador).

```js
chrome.runtime.onMessage.addListener((msg, _sender, sendResponse) => {
  if (msg?.type !== "download") return false;

  chrome.downloads
    .download({ url: msg.url, filename: msg.filename, saveAs: false })
    .then((id) => sendResponse({ ok: true, id }))
    .catch((err) => sendResponse({ ok: false, error: String(err?.message ?? err) }));

  return true; // resposta assíncrona
});
```

```js
const ALPHABET =
  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";

export function shortcodeToId(shortcode) {
  let id = 0n;
  for (const ch of shortcode) {
    const value = ALPHABET.indexOf(ch);
    if (value === -1) throw new Error(`caractere invalido no shortcode: ${ch}`);
    id = id * 64n + BigInt(value);
  }
  return id.toString();
}

export function shortcodeFromUrl(url) {
  const match = String(url).match(/\/(?:p|reel|reels|tv)\/([A-Za-z0-9_-]+)/);
  return match ? match[1] : null;
}

export function filenameFromUrl(url) {
  let pathname;
  try {
    pathname = new URL(url, "https://www.instagram.com").pathname;
  } catch {
    return null;
  }
  const last = pathname.split("/").filter(Boolean).pop() ?? "";
  const clean = last.replace(/[^A-Za-z0-9._-]/g, "_");
  return /\.[A-Za-z0-9]{2,4}$/.test(clean) ? clean : null;
}
```

```js
import { filenameFromUrl } from "./ids.js";

function widest(candidates) {
  if (!Array.isArray(candidates) || candidates.length === 0) return null;
  return candidates.reduce((best, c) => ((c?.width ?? 0) > (best?.width ?? 0) ? c : best));
}

function pickOne(node) {
  const video = widest(node?.video_versions);
  if (video?.url) return { type: "video", url: video.url };

  const image = widest(node?.image_versions2?.candidates);
  if (image?.url) return { type: "image", url: image.url };

  return null;
}

export function collectMedia(item) {
  const nodes = Array.isArray(item?.carousel_media) ? item.carousel_media : [item];
  const media = nodes.map(pickOne).filter(Boolean);
  if (media.length === 0) throw new Error("resposta sem midia utilizavel");
  return media;
}

export function matchIndexByUrl(visibleUrls, list) {
  const visible = new Set(visibleUrls.map(filenameFromUrl).filter(Boolean));
  const index = list.findIndex((m) => visible.has(filenameFromUrl(m.url)));
  return index === -1 ? 0 : index;
}
```

### Task 4: Resolver em camadas

**Files:**
- Create: `src/resolver.js`

**Interfaces:**
- Consumes: `shortcodeToId`, `shortcodeFromUrl` de `src/lib/ids.js`; `collectMedia`, `matchIndexByUrl` de `src/lib/media.js`
- Produces:
  - `resolvePost(permalink: string, visibleUrls: string[]) => Promise<{ list: Array<{type,url}>, index: number }>`
  - `resolveStory(reelId: string, visibleUrls: string[]) => Promise<{ list, index }>`
  - `resolveFromDom(visibleUrls: string[]) => { list, index }` — plano B só para imagem
  - `resolveAuto({ permalink?: string, reelId?: string, visibleUrls: string[] }) => Promise<{ list, index }>` — tenta story, depois post, depois DOM. É o que o content script usa.
  - `IG_APP_ID` (constante exportada)

- [ ] **Step 1: Implementar `src/resolver.js`**

```js
import { shortcodeToId, shortcodeFromUrl } from "./lib/ids.js";
import { collectMedia, matchIndexByUrl } from "./lib/media.js";

export const IG_APP_ID = "936619743392459";

async function apiJson(path) {
  const res = await fetch(`https://www.instagram.com${path}`, {
    credentials: "include",
    headers: { "x-ig-app-id": IG_APP_ID },
  });
  if (!res.ok) throw new Error(`Instagram respondeu ${res.status}`);
  return res.json();
}

export async function resolvePost(permalink, visibleUrls) {
  const shortcode = shortcodeFromUrl(permalink);
  if (!shortcode) throw new Error("nao encontrei o codigo do post");

  const id = shortcodeToId(shortcode);
  const json = await apiJson(`/api/v1/media/${id}/info/`);
  const item = json?.items?.[0];
  if (!item) throw new Error("resposta sem o post");

  const list = collectMedia(item);
  return { list, index: matchIndexByUrl(visibleUrls, list) };
}

export async function resolveStory(reelId, visibleUrls) {
  const json = await apiJson(
    `/api/v1/feed/reels_media/?reel_ids=${encodeURIComponent(reelId)}`
  );
  const reel = json?.reels_media?.[0] ?? json?.reels?.[reelId];
  const items = reel?.items ?? [];
  if (items.length === 0) throw new Error("nenhum story disponivel");

  const list = items.map((item) => collectMedia(item)[0]);
  return { list, index: matchIndexByUrl(visibleUrls, list) };
}

export function resolveFromDom(visibleUrls) {
  const url = visibleUrls.find((u) => /\.(jpe?g|png|webp)/i.test(u));
  if (!url) throw new Error("nao consegui achar o arquivo original");
  return { list: [{ type: "image", url }], index: 0 };
}

export async function resolveAuto({ permalink, reelId, visibleUrls }) {
  try {
    if (reelId) return await resolveStory(reelId, visibleUrls);
    if (permalink) return await resolvePost(permalink, visibleUrls);
  } catch (err) {
    console.warn("[ig-saver] camada 1 falhou:", err.message);
  }
  return resolveFromDom(visibleUrls);
}
```

```js
export function createSaveButton({ onSave, onSaveAll }) {
  const host = document.createElement("div");
  host.dataset.state = "idle";
  const root = host.attachShadow({ mode: "open" });

  const style = document.createElement("style");
  style.textContent = CSS;

  const wrap = document.createElement("div");
  wrap.className = "wrap";
  wrap.innerHTML = `
    <span class="badge hidden"></span>
    <button class="all hidden" type="button">todos</button>
    <button class="save" type="button" title="Salvar">${ICONS.idle}</button>
  `;

  root.append(style, wrap);

  const badge = wrap.querySelector(".badge");
  const allBtn = wrap.querySelector("button.all");
  const saveBtn = wrap.querySelector("button.save");

  const stop = (event) => {
    event.preventDefault();
    event.stopPropagation();
  };
  saveBtn.addEventListener("click", (e) => { stop(e); onSave(); });
  allBtn.addEventListener("click", (e) => { stop(e); onSaveAll(); });

  let resetTimer;
  function setState(state, message = "") {
    clearTimeout(resetTimer);
    host.dataset.state = state;
    saveBtn.innerHTML = ICONS[state] ?? ICONS.idle;
    saveBtn.title = message || (state === "loading" ? "Baixando..." : "Salvar");
    if (state === "done" || state === "error") {
      resetTimer = setTimeout(() => setState("idle"), state === "done" ? 1500 : 4000);
    }
  }

  function setCount(index, total) {
    const many = total > 1;
    badge.classList.toggle("hidden", !many);
    allBtn.classList.toggle("hidden", !many);
    if (many) badge.textContent = `${index + 1}/${total}`;
  }

  function setVisible(visible) {
    wrap.classList.toggle("visible", visible);
  }

  return { host, setState, setCount, setVisible };
}
```

```js
const ATTACHED = "igSaverAttached";

function download(url) {
  const filename = filenameFromUrl(url) ?? `instagram_${Date.now()}.jpg`;
  return new Promise((resolve, reject) => {
    chrome.runtime.sendMessage({ type: "download", url, filename }, (res) => {
      if (chrome.runtime.lastError) return reject(new Error(chrome.runtime.lastError.message));
      if (!res?.ok) return reject(new Error(res?.error ?? "download recusado"));
      resolve(res.id);
    });
  });
}

function visibleUrlsOf(container) {
  const urls = [];
  for (const img of container.querySelectorAll("img")) {
    if (img.currentSrc || img.src) urls.push(img.currentSrc || img.src);
  }
  for (const video of container.querySelectorAll("video")) {
    if (video.poster) urls.push(video.poster);
  }
  return urls;
}

function permalinkOf(container) {
  const link = container.querySelector('a[href*="/p/"], a[href*="/reel/"]');
  if (link) return link.href;
  return /\/(p|reel|reels|tv)\//.test(location.pathname) ? location.href : null;
}

function attach(container) {
  if (container.dataset[ATTACHED]) return;
  container.dataset[ATTACHED] = "1";

  if (getComputedStyle(container).position === "static") {
    container.style.position = "relative";
  }

  const ui = createSaveButton({
    onSave: () => run(false),
    onSaveAll: () => run(true),
  });
  container.append(ui.host);

  container.addEventListener("mouseenter", () => ui.setVisible(true));
  container.addEventListener("mouseleave", () => ui.setVisible(false));

  async function run(all) {
    ui.setState("loading");
    try {
      const { list, index } = await resolveAuto({
        permalink: permalinkOf(container),
        visibleUrls: visibleUrlsOf(container),
      });
      ui.setCount(index, list.length);

      const targets = all ? list : [list[index] ?? list[0]];
      for (const media of targets) {
        await download(media.url);
        if (targets.length > 1) await new Promise((r) => setTimeout(r, 350));
      }
      ui.setState("done");
    } catch (err) {
      console.warn("[ig-saver]", err);
      ui.setState("error", err.message);
    }
  }

  // primeira leitura do contador, sem baixar nada
  resolveAuto({ permalink: permalinkOf(container), visibleUrls: visibleUrlsOf(container) })
    .then(({ list, index }) => ui.setCount(index, list.length))
    .catch(() => {});
}

function scan() {
  for (const article of document.querySelectorAll("article")) attach(article);
}

const observer = new MutationObserver(() => scan());
observer.observe(document.body, { childList: true, subtree: true });
scan();
```
